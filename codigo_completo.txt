USE pasteleriaweb;

CREATE TABLE IF NOT EXISTS rol (
    id_rol     BIGINT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL,
    estado     INT NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS usuario (
    id_usuario BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_rol     BIGINT NOT NULL,
    nombre     VARCHAR(100) NOT NULL,
    apellido   VARCHAR(100) NOT NULL,
    correo     VARCHAR(150) NOT NULL,
    telefono   VARCHAR(20),
    contrasena VARCHAR(255) NOT NULL,
    estado     INT NOT NULL DEFAULT 1,
    CONSTRAINT fk_usuario_rol FOREIGN KEY (id_rol) REFERENCES rol (id_rol)
);

CREATE TABLE IF NOT EXISTS direccion_usuario (
    id_direccion BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_usuario   BIGINT NOT NULL,
    direccion    VARCHAR(255) NOT NULL,
    CONSTRAINT fk_direccion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario)
);

CREATE TABLE IF NOT EXISTS categoria (
    id_categoria BIGINT AUTO_INCREMENT PRIMARY KEY,
    nombre       VARCHAR(100) NOT NULL,
    descripcion  VARCHAR(255),
    estado       INT NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS producto (
    id_producto  BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_categoria BIGINT NOT NULL,
    nombre       VARCHAR(150) NOT NULL,
    descripcion  VARCHAR(500),
    precio       DOUBLE NOT NULL,
    imagen       VARCHAR(255),
    stock        INT NOT NULL DEFAULT 0,
    estado       INT NOT NULL DEFAULT 1,
    CONSTRAINT fk_producto_categoria FOREIGN KEY (id_categoria) REFERENCES categoria (id_categoria)
);

CREATE TABLE IF NOT EXISTS pedido (
    id_pedido         BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_usuario        BIGINT NOT NULL,
    id_direccion      BIGINT NOT NULL,
    fecha_pedido      DATETIME,
    fecha_entrega     DATETIME,
    modalidad_entrega VARCHAR(30),
    estado            VARCHAR(30),
    total             DECIMAL(10,2),
    observacion       VARCHAR(500),
    CONSTRAINT fk_pedido_usuario   FOREIGN KEY (id_usuario)   REFERENCES usuario (id_usuario),
    CONSTRAINT fk_pedido_direccion FOREIGN KEY (id_direccion) REFERENCES direccion_usuario (id_direccion)
);

CREATE TABLE IF NOT EXISTS detallepedido (
    id_detalle      BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_pedido       BIGINT NOT NULL,
    id_producto     BIGINT NOT NULL,
    cantidad        INT NOT NULL,
    precio_unitario DECIMAL(10,2),
    subtotal        DECIMAL(10,2),
    CONSTRAINT fk_detalle_pedido   FOREIGN KEY (id_pedido)   REFERENCES pedido (id_pedido),
    CONSTRAINT fk_detalle_producto FOREIGN KEY (id_producto) REFERENCES producto (id_producto)
);

CREATE TABLE IF NOT EXISTS entrega (
    id_entrega       BIGINT AUTO_INCREMENT PRIMARY KEY,
    id_pedido        BIGINT NOT NULL UNIQUE,
    id_motorizado    BIGINT NOT NULL,
    fecha_asignacion DATETIME,
    fecha_entrega    DATETIME,
    estado           INT NOT NULL DEFAULT 0,
    costo_delivery   DECIMAL(10,2),
    observacion      VARCHAR(500),
    CONSTRAINT fk_entrega_pedido     FOREIGN KEY (id_pedido)     REFERENCES pedido (id_pedido),
    CONSTRAINT fk_entrega_motorizado FOREIGN KEY (id_motorizado) REFERENCES usuario (id_usuario)
);

    }

    public List<Usuario> listarporNombre(String nombre) {
        return usuarioR.findByNombre(nombre);

    public Usuario listarID(Long id) {
        return usuarioR.findById(id).orElse(null);
    }

    public Usuario guardar(Usuario usuario) {
        return usuarioR.save(usuario);
    }

    public Usuario actualizar(Long id, Usuario detalles) {
        Optional<Usuario> optional = usuarioR.findById(id);
        if (optional.isPresent()) {
            Usuario existente = optional.get();
            existente.setRol(detalles.getRol());
            existente.setNombre(detalles.getNombre());
            existente.setApellido(detalles.getApellido());
            existente.setCorreo(detalles.getCorreo());
            existente.setTelefono(detalles.getTelefono());
            existente.setContrasena(detalles.getContrasena());
            existente.setEstado(detalles.getEstado());
            return usuarioR.save(existente);
        }
        return null;
    }

    public boolean eliminar(Long id) {
        if (usuarioR.existsById(id)) {
        	usuarioR.deleteById(id);
            return true;
        }
        return false;
    }

async function llamar(ruta, metodo, cuerpo) {
  const opciones = { method: metodo };
  if (cuerpo instanceof FormData) {
    opciones.body = cuerpo;
  } else if (cuerpo !== undefined) {
    opciones.headers = { "Content-Type": "application/json" };
    opciones.body = JSON.stringify(cuerpo);
  }
  const respuesta = await fetch(ruta, opciones);
  const texto = await respuesta.text();
  let datos = texto;
  try { datos = JSON.parse(texto); } catch (e) {  }
  if (!respuesta.ok) {
    throw new Error(typeof datos === "string" && datos ? datos : "Error " + respuesta.status);
  }
  return datos;
}

  const error = fondo.querySelector(".error-form");
  const cerrar = () => fondo.remove();

  fondo.querySelectorAll("[data-cerrar]").forEach((b) => b.addEventListener("click", cerrar));
  fondo.addEventListener("mousedown", (e) => { if (e.target === fondo) cerrar(); });

  formulario.addEventListener("submit", async (e) => {
    e.preventDefault();
    const guardar = formulario.querySelector('[type="submit"]');
    error.hidden = true;
    guardar.disabled = true;
    try {
      await alGuardar(formulario);
      cerrar();
    } catch (err) {
      error.textContent = err.message;
      error.hidden = false;
      guardar.disabled = false;
    }
  });

  const primero = formulario.querySelector("input, select, textarea");
  if (primero) primero.focus();

function pedidosPorDia(pedidos) {

    const d = new Date();
    d.setHours(0, 0, 0, 0);

}

async function enviar(ruta, metodo, cuerpo) {
  const respuesta = await fetch(ruta, {
    method: metodo,
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(cuerpo),
  });
  const texto = await respuesta.text();
  if (!respuesta.ok) {
    let mensaje = texto || "Error " + respuesta.status;
    try {
      const interpretado = JSON.parse(texto); 
      if (typeof interpretado === "string") mensaje = interpretado;
    } catch (e) {  }
    throw new Error(mensaje);
  }
  return texto ? JSON.parse(texto) : null;
}

    } catch (e) {
      return [];
    }
  },
  guardar(lista) {
    try {
      localStorage.setItem(CLAVE_CARRITO, JSON.stringify(lista));
    } catch (e) {

    }
  },
  agregar(idProducto, cantidad) {
    const lista = this.leer();
    const existente = lista.find((i) => i.id === idProducto);
    if (existente) existente.cantidad += cantidad;
    else lista.push({ id: idProducto, cantidad: cantidad });
    this.guardar(lista);
    this.actualizarInsignia();
  },
  cambiarCantidad(idProducto, cantidad) {
    const lista = this.leer();
    const item = lista.find((i) => i.id === idProducto);
    if (item) item.cantidad = cantidad;
    this.guardar(lista);
    this.actualizarInsignia();
  },
  quitar(idProducto) {
    this.guardar(this.leer().filter((i) => i.id !== idProducto));
    this.actualizarInsignia();
  },
  vaciar() {
    this.guardar([]);
    this.actualizarInsignia();
  },
  totalUnidades() {
    return this.leer().reduce((suma, i) => suma + i.cantidad, 0);
  },
  actualizarInsignia() {
    const insignia = document.querySelector(".nav-carrito .n");
    if (!insignia) return;
    const total = this.totalUnidades();
    insignia.textContent = total;

  },
